const EnquiryModal = ({ projectName, onClose }: { projectName: string, onClose: () => void }) => {
  const [loading, setLoading] = useState(false);
  const [submitted, setSubmitted] = useState(false);

  const handleWhatsAppChat = () => {
    const message = encodeURIComponent(`Hi Gomatha Realtors, I am interested in "${projectName}". Please share more details.`);
    window.open(`https://wa.me/${COMPANY_DETAILS.whatsapp}?text=${message}`, '_blank');
  };

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    setLoading(true);
    const formData = new FormData(e.currentTarget);
    const data = {
      projectName: projectName,
      name: formData.get('name'),
      phone: formData.get('phone'),
      message: formData.get('message'),
    };

    try {
      const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbxE1OzO9XbH2Qrf6PKmHRFDI7woJVH_7M2OohSuL654WpBrNVV5HnRD_xnj19T_7hJV/exec'; 
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors', 
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data),
      });
      setSubmitted(true);
      setTimeout(() => onClose(), 3000);
    } catch (error) {
      alert("Submission failed. Please use WhatsApp.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="fixed inset-0 z-[110] flex items-center justify-center p-4">
      <div className="fixed inset-0 bg-navy/95 backdrop-blur-md" onClick={onClose} />
      <div className="relative w-full max-w-xl bg-white rounded-[40px] shadow-2xl overflow-hidden animate-fade-in-up p-8 md:p-14">
        {submitted ? (
          <div className="text-center py-10 animate-fade-in-up">
            <CheckCircle2 size={60} className="text-green-500 mx-auto mb-4" />
            <h3 className="text-3xl font-bold text-navy">Inquiry Received!</h3>
            <p className="text-gray-500 mt-2">Data synced to Google Sheets. Closing...</p>
          </div>
        ) : (
          <>
            <div className="flex justify-between items-start mb-8">
              <div>
                <span className="text-gold font-black text-[10px] uppercase tracking-widest mb-1 block">Property Inquiry</span>
                <h3 className="text-2xl font-bold text-navy">{projectName}</h3>
              </div>
              <button onClick={onClose} className="text-navy p-2 hover:bg-gray-100 rounded-full transition"><X size={24} /></button>
            </div>
            
            <button onClick={handleWhatsAppChat} className="w-full mb-8 py-4 bg-[#25D366] text-white rounded-2xl font-bold flex items-center justify-center gap-3 hover:scale-[1.02] transition">
              <MessageCircle size={20} /> Chat on WhatsApp
            </button>

            <form className="space-y-4" onSubmit={handleSubmit}>
              <div className="grid md:grid-cols-2 gap-4">
                <input name="name" required className="w-full p-4 bg-gray-50 border border-gray-100 rounded-xl outline-none focus:ring-2 focus:ring-gold" placeholder="Your Name" />
                <input name="phone" required type="tel" className="w-full p-4 bg-gray-50 border border-gray-100 rounded-xl outline-none focus:ring-2 focus:ring-gold" placeholder="Contact Number" />
              </div>
              <textarea name="message" className="w-full p-4 bg-gray-50 border border-gray-100 rounded-xl outline-none focus:ring-2 focus:ring-gold" rows={3} placeholder="Tell us your requirements..."></textarea>
              <button disabled={loading} type="submit" className="w-full py-5 bg-navy text-gold rounded-xl font-bold uppercase tracking-widest flex items-center justify-center gap-3 hover:bg-navy-light transition shadow-lg">
                {loading ? <Loader2 className="animate-spin" /> : "Submit to Sheet"}
              </button>
            </form>
          </>
        )}
      </div>
    </div>
  );
};
